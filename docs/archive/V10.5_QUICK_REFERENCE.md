## 📋 **أمثلة عملية:**

### **مثال 1: تحميل ملف تنفيذ**

```typescript
import { storageCache } from '@/app/utils/storageCache';
import { ExecutionToasts } from '@/app/utils/toastMessages';
import { handleAsyncError } from '@/app/utils/errorHandler';
import { ExecutionDashboardSkeleton } from '@/app/components/ui/Skeleton';

const [isLoading, setIsLoading] = useState(true);

useEffect(() => {
  const loadFile = async () => {
    setIsLoading(true);
    
    const result = await handleAsyncError(
      async () => {
        // محاولة القراءة من Cache أولاً
        let file = storageCache.get(`execution_${executionId}`);
        
        // إذا لم يكن موجوداً، جلب من localStorage
        if (!file) {
          const files = storageCache.get('execution_files') || [];
          file = files.find(f => f.id === executionId);
          
          if (file) {
            // حفظ في Cache للمرة القادمة
            storageCache.set(`execution_${executionId}`, file);
          }
        }
        
        return file;
      },
      'فشل تحميل ملف التنفيذ'
    );
    
    if (result) {
      setExecutionData(result);
      ExecutionToasts.info.calculationUpdated();
    }
    
    setIsLoading(false);
  };
  
  loadFile();
}, [executionId]);

if (isLoading) {
  return <ExecutionDashboardSkeleton />;
}
```

### **مثال 2: حفظ دفعة**

```typescript
import { storageCache } from '@/app/utils/storageCache';
import { ExecutionToasts } from '@/app/utils/toastMessages';
import { handleAsyncError } from '@/app/utils/errorHandler';

const handlePayment = async (amount: number) => {
  if (!amount || amount <= 0) {
    ExecutionToasts.error.invalidAmount();
    return;
  }
  
  const result = await handleAsyncError(
    async () => {
      // تحديث البيانات
      const updatedFile = {
        ...executionData,
        paidDebt: (executionData.paidDebt || 0) + amount,
        lastPaymentDate: new Date().toISOString()
      };
      
      // حفظ في Cache
      storageCache.set(`execution_${executionData.id}`, updatedFile);
      
      // تحديث القائمة
      const files = storageCache.get('execution_files') || [];
      const updatedFiles = files.map(f => 
        f.id === executionData.id ? updatedFile : f
      );
      storageCache.set('execution_files', updatedFiles);
      
      return updatedFile;
    },
    'فشل حفظ الدفعة'
  );
  
  if (result) {
    setExecutionData(result);
    ExecutionToasts.success.paymentRecorded(amount);
  }
};
```

### **مثال 3: معالجة أخطاء**

```typescript
import { logErrorWithContext } from '@/app/utils/errorHandler';
import { ExecutionToasts } from '@/app/utils/toastMessages';

const handleDangerousOperation = () => {
  try {
    // عملية خطرة
    const result = complexCalculation();
    
    if (!result.isValid) {
      throw new Error('النتيجة غير صحيحة');
    }
    
    ExecutionToasts.success.operationComplete();
    
  } catch (error) {
    // تسجيل مع سياق كامل
    logErrorWithContext('DangerousOperation', error, {
      userId: currentUser?.id,
      fileId: executionData?.id,
      timestamp: new Date().toISOString()
    });
    
    // إظهار رسالة للمستخدم
    ExecutionToasts.error.operationFailed();
  }
};
```
